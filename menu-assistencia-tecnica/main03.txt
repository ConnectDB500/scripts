# main.py
from menu.interfaces.menu import menu
from menu.linux.index import run as linux_run

def main():
    # Mostra menu principal (menu[0])
    for item in menu[0]:
        print(item)

    escolha = input("\nDigite sua escolha: ").strip()

    if escolha != "1":
        print("Programa encerrado.")
        return

    # Mostra submenu (menu[1])
    for item in menu[1]:
        print(item)

    escolha2 = input("\nDigite sua escolha: ").strip()

    if escolha2 == "2":
        linux_run()
    else:
        print("Programa encerrado.")

if __name__ == "__main__":
    main()
