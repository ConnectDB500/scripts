"""
/projeto/main.py

Ponto de entrada simples:
- carrega menu (tenta menu.interfaces.menu ou menu.interface.menu)
- exibe menu[0]; se escolha == 1 exibe menu[1]
- se usuário escolher '2' (Linux) chama menu.linux.index.run()
"""

import importlib
import importlib.util
import os
import sys

def load_menu():
    """Tenta importar menu a partir de pacotes conhecidos ou arquivo."""
    candidates = [
        "menu.interfaces.menu",
        "menu.interface.menu",  # variação caso você tenha usado 'interface' sem s
    ]
    for name in candidates:
        try:
            mod = importlib.import_module(name)
            if hasattr(mod, "menu"):
                return mod.menu
        except ModuleNotFoundError:
            continue

    # fallback: carregar por caminho de arquivo
    possible_paths = [
        os.path.join("menu", "interfaces", "menu.py"),
        os.path.join("menu", "interface", "menu.py")
    ]
    for path in possible_paths:
        if os.path.exists(path):
            spec = importlib.util.spec_from_file_location("menu_file", path)
            module = importlib.util.module_from_spec(spec)
            spec.loader.exec_module(module)
            if hasattr(module, "menu"):
                return module.menu
    raise RuntimeError("Arquivo menu.py não encontrado em menu/interfaces ou menu/interface, ou 'menu' não está definido.")

def call_linux_index():
    """
    Tenta importar menu.linux.index e chamar run().
    Se não conseguir importar como pacote, carrega por caminho.
    """
    try:
        mod = importlib.import_module("menu.linux.index")
    except ModuleNotFoundError:
        # tentar via caminho relativo (menu/linux/index.py)
        path = os.path.join(os.path.dirname(__file__), "menu", "linux", "index.py")
        if not os.path.exists(path):
            print("Não foi possível localizar menu/linux/index.py")
            return
        spec = importlib.util.spec_from_file_location("menu_linux_index", path)
        mod = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(mod)

    # função padrão nomeada run() ou main()
    if hasattr(mod, "run") and callable(mod.run):
        mod.run()
    elif hasattr(mod, "main") and callable(mod.main):
        mod.main()
    else:
        print("O módulo menu.linux.index não expõe run() nem main().")

def main():
    menu = load_menu()
    # Mostrar menu[0]
    print()
    for line in menu[0]:
        print(line)
    escolha = input("\nDigite sua escolha: ").strip()
    try:
        num = int(escolha)
    except ValueError:
        print("Entrada inválida. Encerrando.")
        return

    if num != 1:
        print("Encerrando o programa.")
        return

    # mostra o submenu menu[1]
    print()
    for line in menu[1]:
        print(line)
    escolha2 = input("\nDigite sua escolha: ").strip()
    try:
        sub = int(escolha2)
    except ValueError:
        print("Entrada inválida. Encerrando.")
        return

    # conforme seu pedido: se for 2 -> linux; senao encerra
    if sub == 2:
        call_linux_index()
    else:
        print("Encerrando o programa.")
        return

if __name__ == "__main__":
    main()
