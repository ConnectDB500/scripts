"""
/projeto/menu/linux/index.py

Interface de navegação para as funcionalidades Linux.

Fluxo:
- carrega menu de categorias a partir de menu.interfaces.menu_linux
- mapeia opção escolhida para um módulo de dicionário (ex: 'checagem', 'diagnostico', ...)
- carrega esse dicionário e mostra suas opções
- ao escolher uma opção, pega item["linux"] (ex: "linux.backup_drivers")
  e tenta:
    1) importar como módulo Python (importlib) e chamar index()/main()/run()
    2) se falhar, localizar pasta física PROJECT_ROOT/linux/backup_drivers/
       e executar index.py (python) ou index.sh (bash) dentro dela.
"""

import importlib
import importlib.util
import os
import subprocess
import sys
from typing import Optional

PROJECT_ROOT = os.path.abspath(os.path.join(os.path.dirname(__file__), "..", ".."))

# Mapeamento: escolha de categoria -> módulo com dicionário em menu/interfaces/<nome>.py
CATEGORY_MAP = {
    "1": "checagem",
    "2": "diagnostico",
    "3": "otimizacao",
    "4": "limpeza",
    "5": "backup",
    "6": "voltar",
}

def load_menu_linux():
    """Carrega a lista menu de menu/interfaces/menu_linux.py (tenta pacote e caminho)."""
    try:
        mod = importlib.import_module("menu.interfaces.menu_linux")
        return mod.menu
    except ModuleNotFoundError:
        # fallback para arquivo
        path = os.path.join(PROJECT_ROOT, "menu", "interfaces", "menu_linux.py")
        if os.path.exists(path):
            spec = importlib.util.spec_from_file_location("menu_linux", path)
            m = importlib.util.module_from_spec(spec)
            spec.loader.exec_module(m)
            return m.menu
    raise RuntimeError("menu_linux.py não encontrado em menu/interfaces")

def load_dict_module(dict_name: str):
    """
    Importa menu.interfaces.<dict_name> e retorna o dicionário interno.
    Ex: dict_name='checagem' -> importa menu.interfaces.checagem e retorna o objeto 'checagem'.
    """
    full_mod = f"menu.interfaces.{dict_name}"
    try:
        mod = importlib.import_module(full_mod)
    except ModuleNotFoundError:
        # tentar pelo caminho do arquivo
        path = os.path.join(PROJECT_ROOT, "menu", "interfaces", f"{dict_name}.py")
        if not os.path.exists(path):
            raise
        spec = importlib.util.spec_from_file_location(full_mod, path)
        mod = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(mod)

    if not hasattr(mod, dict_name):
        raise AttributeError(f"O módulo {full_mod} não define a variável '{dict_name}'")
    return getattr(mod, dict_name)

def try_import_and_call(module_path: str) -> bool:
    """
    Tenta importar module_path (ex: 'linux.backup_drivers') e executar uma função.
    Procura por: index(), main(), run() como entry-points.
    Retorna True se executou algo, False caso contrário.
    """
    try:
        mod = importlib.import_module(module_path)
    except ModuleNotFoundError:
        return False

    for func_name in ("index", "main", "run"):
        func = getattr(mod, func_name, None)
        if callable(func):
            try:
                func()
                return True
            except Exception as e:
                print(f"Erro ao executar {module_path}.{func_name}(): {e}")
                return True  # já importou, consideramos que tentou executar
    # tentativa: se é um package, tentar module_path.index submodule
    try:
        sub = importlib.import_module(module_path + ".index")
        if hasattr(sub, "index") and callable(sub.index):
            sub.index()
            return True
    except Exception:
        pass

    # importou mas não achou função executável
    return False

def try_exec_by_path(module_path: str) -> bool:
    """
    Converte 'linux.backup_drivers' -> <PROJECT_ROOT>/linux/backup_drivers and tenta:
      - executar index.py (python) se existir
      - executar index.sh (bash) se existir
      - executar <module>.py se existir (ex: linux/backup_drivers.py)
    Retorna True se encontrou e executou algo.
    """
    parts = module_path.split(".")
    path_dir = os.path.join(PROJECT_ROOT, *parts)  # e.g. /projeto/linux/backup_drivers

    # Caso exista um arquivo python direto: linux/backup_drivers.py (menos comum)
    py_file = path_dir + ".py"
    if os.path.isfile(py_file):
        print(f"Executando script Python: {py_file}")
        subprocess.run([sys.executable, py_file], check=False)
        return True

    if os.path.isdir(path_dir):
        index_py = os.path.join(path_dir, "index.py")
        index_sh = os.path.join(path_dir, "index.sh")

        if os.path.isfile(index_py):
            print(f"Executando index.py em {path_dir}")
            subprocess.run([sys.executable, index_py], cwd=path_dir, check=False)
            return True
        if os.path.isfile(index_sh):
            print(f"Executando index.sh em {path_dir}")
            subprocess.run(["bash", index_sh], cwd=path_dir, check=False)
            return True

    # nada encontrado
    return False

def execute_linux_target(module_path: str):
    """
    Estratégia unificada para executar o 'module_path' definido no dicionário.
    Tenta import + call, depois tenta executar por caminho.
    """
    print(f"Tentando executar: {module_path}")

    # 1) tentar importar e executar função
    try:
        ok = try_import_and_call(module_path)
        if ok:
            return
    except Exception as e:
        print(f"Erro durante import/exec: {e}")

    # 2) tentar executar por caminho (index.py / index.sh)
    try:
        ok2 = try_exec_by_path(module_path)
        if ok2:
            return
    except Exception as e:
        print(f"Erro durante execução por caminho: {e}")

    print(f"Não foi possível executar {module_path}. Verifique se o módulo/script existe.")

def show_dict_menu_and_execute(dict_obj: dict):
    """
    Recebe o dicionário (ex: checagem) e exibe as opções,
    pergunta a escolha e executa a rota 'linux' correspondente.
    """
    while True:
        print()
        keys = sorted(dict_obj.keys(), key=lambda k: int(k))
        for k in keys:
            print(dict_obj[k]["descricao"])
        escolha = input("\nEscolha uma opção (ou digite 'b' para voltar): ").strip()
        if escolha.lower() in ("b", "voltar"):
            return
        if escolha not in dict_obj:
            print("Opção inválida. Tente novamente.")
            continue
        item = dict_obj[escolha]
        # Se não tiver 'linux', consideramos que é 'Voltar' / ou não implementado
        if "linux" not in item:
            print("Opção de voltar ou não implementada para Linux. Voltando...")
            return
        execute_linux_target(item["linux"])
        # depois de executar, volta a mostrar as opções desse dicionário

def run():
    """Ponto de entrada para o menu Linux."""
    # 1) carregar menu_linux
    menu_list = load_menu_linux()
    while True:
        print("\nMenu: Otimização / Diagnóstico / Limpeza (Linux)")
        for line in menu_list:
            print(line)
        escolha = input("\nEscolha uma opção: ").strip()
        if escolha not in CATEGORY_MAP:
            print("Opção inválida. Tente novamente.")
            continue
        if escolha == "6":  # voltar
            print("Voltando ao menu anterior...")
            break

        dict_name = CATEGORY_MAP[escolha]
        try:
            menu_dict = load_dict_module(dict_name)
        except Exception as e:
            print(f"Erro ao carregar dicionário '{dict_name}': {e}")
            continue

        # mostrar opções desse dicionário e executar
        show_dict_menu_and_execute(menu_dict)

if __name__ == "__main__":
    run()
