# index.py
import importlib
import subprocess
from menu.interfaces import menu_linux  # aqui está a lista simples com as categorias Linux

def run():
    # Mostra opções principais Linux
    for item in menu_linux.menu:
        print(item)

    escolha = input("\nDigite sua escolha: ").strip()

    if escolha == "6":  # Voltar
        return

    # Mapeamento simples das opções para os nomes dos módulos/dicionários
    modulos = {
        "1": "checagem",
        "2": "diagnostico",
        "3": "otimizacao",
        "4": "limpeza",
        "5": "backup"
    }

    if escolha not in modulos:
        print("Opção inválida.")
        return

    # Importa o dicionário correspondente ao submenu
    modulo = importlib.import_module(f"menu.interfaces.{modulos[escolha]}")
    dados = getattr(modulo, modulos[escolha])

    # Mostra as opções do submenu
    for chave, valor in dados.items():
        print(valor["descricao"])

    escolha_sub = input("\nDigite sua escolha: ").strip()

    if escolha_sub not in dados or "linux" not in dados[escolha_sub]:
        print("Voltando...")
        return

    # Monta o caminho do script .sh para execução
    caminho_script = dados[escolha_sub]["linux"].replace(".", "/") + ".sh"

    print(f"Executando {caminho_script}...\n")

    subprocess.run(["bash", caminho_script])
