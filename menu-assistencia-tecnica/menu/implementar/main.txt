import os
import subprocess
from menu.interface import menu  # Exemplo, ajuste conforme seu código
from menu.layout import topo, rodape

def listar_funcoes(sistema):
    pasta = sistema
    funcoes = [d for d in os.listdir(pasta) if os.path.isdir(os.path.join(pasta, d))]
    return funcoes

def exibir_menu_opcoes(opcoes):
    for i, op in enumerate(opcoes, 1):
        print(f"{i} - {op}")
    escolha = int(input("Escolha uma opção: "))
    return opcoes[escolha - 1]

def main():
    topo("AUXILIAR SUPORTE DE TI")
    
    sistemas = ["windows", "linux"]
    print("Escolha o sistema operacional:")
    so = exibir_menu_opcoes(sistemas)

    funcoes = listar_funcoes(so)
    print(f"Escolha a função para {so}:")
    funcao = exibir_menu_opcoes(funcoes)

    extensao = ".ps1" if so == "windows" else ".sh"
    caminho_script = os.path.join(so, funcao, f"index{extensao}")

    print(f"Executando {caminho_script}...")
    if so == "windows":
        subprocess.run(["powershell.exe", "-File", caminho_script])
    else:
        subprocess.run(["bash", caminho_script])
    
    rodape()

if __name__ == "__main__":
    main()
