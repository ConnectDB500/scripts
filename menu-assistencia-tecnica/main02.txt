# /projeto/menu/linux/index.py
import importlib
import os
import subprocess
from menu.layout.menu_interface import topo, rodape

def run():
    # Importa menu_linux
    try:
        from menu.interfaces import menu_linux
    except ImportError:
        print("Erro: não foi possível carregar menu_linux.py")
        return
    
    topo("MENU LINUX")

    # Mostra opções principais do Linux
    for item in menu_linux.menu:
        print(item)

    escolha = input("\nDigite sua escolha: ").strip()
    if not escolha.isdigit():
        print("Entrada inválida.")
        return

    escolha = int(escolha)
    if escolha == len(menu_linux.menu):  # Último item = Voltar
        return

    # Mapeia escolha para nome de arquivo de dicionário
    arquivos_map = {
        1: "checagem",
        2: "diagnostico",
        3: "otimizacao",
        4: "limpeza",
        5: "backup"
    }

    if escolha not in arquivos_map:
        print("Opção inválida.")
        return

    modulo_nome = arquivos_map[escolha]

    try:
        modulo = importlib.import_module(f"menu.interfaces.{modulo_nome}")
    except ImportError:
        print(f"Erro: não foi possível carregar {modulo_nome}.py")
        return

    # Mostra opções do dicionário
    opcoes = getattr(modulo, modulo_nome)
    for chave, dados in opcoes.items():
        print(dados["descricao"])

    sub_escolha = input("\nDigite sua escolha: ").strip()
    if sub_escolha not in opcoes:
        print("Opção inválida.")
        return

    rota = opcoes[sub_escolha].get("linux")
    if not rota:
        print("Voltando...")
        return

    # Converte "linux.nome_pasta" para caminho absoluto
    pasta_script = rota.split(".")[1]  # pega a parte depois de 'linux.'
    caminho_script = os.path.join("linux", pasta_script, "index.sh")

    if not os.path.exists(caminho_script):
        print(f"Script não encontrado: {caminho_script}")
        return

    print(f"\nExecutando {caminho_script}...\n")
    subprocess.run(["bash", caminho_script])

    rodape()
